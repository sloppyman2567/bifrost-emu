# Column format (whitespace separated, '#' comments, blank lines ignored):
#     NAME   LIBS   ARGS   RET   POLICY   SIZE
#
# LIBS    : lib family the symbol is registered for. One row = the symbol is
#           registered under every soname of that family:
#             GL   -> libGL.so, libGL.so.1
#             GLES -> libGLESv2.so, libGLESv2.so.2
#             EGL  -> libEGL.so, libEGL.so.1
#             SDL  -> libSDL2.so, libSDL2-2.0.so.0
#             GLFW -> libglfw.so.3, libglfw.so
# ARGS    : AAPCS64 argument kind string, one char per parameter in order:
#             i  integer/enum register (x-reg), read as raw uint64
#             f  IEEE-754 binary32 register (v-reg) — only used when the
#                whole call is float-only (all f) or mixed (i* + f*)
#             p  pointer argument, translated guest→host with writeback
#             z  pointer argument like p, but bounced at a computed size
#                instead of the default 64 KiB (see SIZE column)
#           '-' means no arguments. Pointer *positions* drive the pointer_args
#           bitmask; the total count only matters when it exceeds 8 (stack
#           args at SP).
# RET     : '-' = plain (void/int, host ret written to x0)
#           str = host returns const char* -> copy into the guest string cache
#           f/d = binary32/binary64 result returned in guest v0
# POLICY  : named dispatch behavior (one switch arm each, no name compares):
#             -              default generic path
#             SDL_SURFACE    bridge guest surfaces, formats, palettes and pixels
#             SHADER_SOURCE  glShaderSource nested string-array marshalling
#             QUERY          GL state queries intercepted by GLStateTracker
#             GET_PROC       *GetProcAddress: resolve guest name -> trampoline
#             TRACK_TEX      SDL_CreateTexture: record size for PITCH_H bounce
#             UNTRACK_TEX    SDL_DestroyTexture: drop recorded size
#             VA_PTR         vertex-attrib pointer: translate arg5 iff no VBO
#                            bound (array buffer), else leave as byte offset
#             EL_PTR         glDrawElements: translate arg3 iff no element
#                            array buffer bound, else leave as byte offset
#             EL_PTR_ARRAY   glMultiDrawElements(*BaseVertex): bounce the
#                            outer count/indices arrays; each inner indices
#                            element is a byte offset iff an element array
#                            buffer is bound, else a translated client ptr
#             INDIRECT_PTR   glDraw*Indirect: the indirect struct arg is a
#                            byte offset iff DRAW_INDIRECT_BUFFER is bound,
#                            else a translated client struct
#             PRESENT        frame-present/swap: BIFROST_FRAME_TRACE counter
#             ERROR_CB       glfwSetErrorCallback (global, single arg): store
#                            the guest cb; deliver after GLFW_POLL with dedup
#             KEY_CB         glfwSetKeyCallback (window, cb): store + deliver
#             MOUSE_CB       glfwSetMouseButtonCallback (window, cb)
#             CURSOR_CB      glfwSetCursorPosCallback (window, cb)
#             FRAMEBUFFER_CB glfwSetFramebufferSizeCallback (window, cb)
#             WINDOW_SIZE_CB glfwSetWindowSizeCallback (window, cb)
#             FOCUS_CB       glfwSetWindowFocusCallback (window, cb)
#             GLFW_POLL      glfwPollEvents/glfwWaitEvents: after the host
#                            call, deliver all stored *_CB callbacks whose
#                            value changed since the last poll
#             GLFW_CREATE    glfwCreateWindow: create the host window, then
#                            compensate for the host content scale (HiDPI)
#                            so the PHYSICAL framebuffer is exactly the
#                            size the guest requested (1280x720 stays
#                            1280x720 instead of 2560x1440 on 2x displays)
#             GL_DEBUG_CB  glDebugMessageCallback(cb, user): store the
#                            guest cb + param, pass NULL to the host (a
#                            guest address handed to host GL would crash
#                            on first driver message). delivery is a
#                            future milestone, like vkMapMemory was.
# SIZE    : bounce-size policy for the z argument(s):
#             -       default 64 KiB
#             arg1    size = guest arg1 (glBufferData: data is arg2)
#             arg2    size = guest arg2 (glBufferSubData: data is arg3)
#             TEX2D   size = w*h*bpp from args 3,4,6,7 (glTexImage2D)
#             TEXSUB  size = w*h*bpp from args 4,5,6,7 (glTexSubImage2D:
#                     target, level, xoffset, yoffset, width, height,
#                     format, type, pixels — args 3 is yoffset, NOT width)
#             PITCH_H size = pitch(arg3) * recorded texture height
#
# AAPCS64 scalar floats travel in v-registers, independently of integer
# arguments in x-registers. Mixed rows currently support integer arguments
# before float arguments; special-case any reversed mixed signature.
